cmake_minimum_required(VERSION 3.24)
project(WindowsWorkspaceTools LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The existing macOS setup uses macOS/Setup.sh; this target is for Windows.")
endif()
if(NOT CMAKE_BINARY_DIR STREQUAL "${CMAKE_SOURCE_DIR}/build")
    message(FATAL_ERROR "Use the repository build/ directory.")
endif()
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(WindowsRuntimeFixture WIN32 Windows/tests/fixture.cpp)
target_compile_features(WindowsRuntimeFixture PRIVATE cxx_std_23)
target_link_libraries(WindowsRuntimeFixture PRIVATE user32)
if(MINGW)
    target_link_options(WindowsRuntimeFixture PRIVATE -municode)
endif()
add_test(NAME Windows.NativeRunner
    COMMAND ${CMAKE_COMMAND} -E env "WINDOWS_RUNTIME_FIXTURE=$<TARGET_FILE:WindowsRuntimeFixture>"
    ${Python3_EXECUTABLE} -B -X utf8 -m unittest discover -s ${CMAKE_SOURCE_DIR}/Windows/tests -v)
